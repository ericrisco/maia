<!-- buit: temes/institucions/nacionalitat-i-residencia/registre-residencia-andorra-franca-1893#g2 · estat: obert -->

família.

<!-- buit: temes/institucions/nacionalitat-i-residencia/registre-residencia-andorra-franca-1893#g4 · estat: obert -->
